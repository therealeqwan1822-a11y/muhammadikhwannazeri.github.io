1. System Architecture & FSM Design

The system architecture is separated into three core hardware modules: Button Debouncer & Edge Detector, Main FSM Controller, and Grid Evaluator Logic.

  • Debounce Stage: Converts tactile switch bounce (~10–20 ms noise) into clean single-clock pulse triggers via a 100 Hz scaled clock divider.
  • FSM Controller: Sequentially transitions between IDLE, WAIT_PLAYER_X, EVAL_X, WAIT_PLAYER_O, EVAL_O, and GAME_OVER states.
  • Combinational Evaluator: Concurrently computes 8 possible win lines (3 horizontal rows, 3 vertical columns, 2 diagonals) in less than 10 ns to evaluate victory conditions instantly after every valid move.

-- ====================================================================
-- Project: Tic-Tac-Toe Game Engine on Intel DE2-115 FPGA
-- Author: Muhammad Ikhwan Bin Nazeri
-- Target: Intel Cyclone IV EP4CE115F29C7
-- Language: VHDL-2008
-- ====================================================================

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity tictactoe_fsm is
    Port (
        clk        : in  STD_LOGIC;                    -- 50 MHz On-Board Oscillator
        reset_n    : in  STD_LOGIC;                    -- Active-Low Synchronous Reset
        move_btn   : in  STD_LOGIC_VECTOR(8 downto 0); -- 3x3 Grid Key Inputs
        led_player : out STD_LOGIC_VECTOR(8 downto 0); -- Red LEDs (Player X)
        led_comp   : out STD_LOGIC_VECTOR(8 downto 0); -- Green LEDs (Player O)
        win_state  : out STD_LOGIC_VECTOR(1 downto 0)  -- "01" = X Wins, "10" = O Wins, "11" = Draw
    );
end tictactoe_fsm;

architecture Behavioral of tictactoe_fsm is

    -- FSM State Definitions
    type state_type is (IDLE, WAIT_MOVE_X, EVAL_X, WAIT_MOVE_O, EVAL_O, CHECK_DRAW, GAME_OVER);
    signal current_state, next_state : state_type;

    -- Grid Representations
    signal grid_x, grid_o : STD_LOGIC_VECTOR(8 downto 0) := (others => '0');

begin

    -- State Register Process
    SYNC_PROC: process(clk, reset_n)
    begin
        if reset_n = '0' then
            current_state <= IDLE;
            grid_x <= (others => '0');
            grid_o <= (others => '0');
        elsif rising_edge(clk) then
            current_state <= next_state;
        end if;
    end process SYNC_PROC;

    -- Output Pin Assignments
    led_player <= grid_x;
    led_comp   <= grid_o;

end Behavioral;
